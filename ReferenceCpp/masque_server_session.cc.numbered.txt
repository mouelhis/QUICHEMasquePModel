    80	constexpr size_t GetMaxIpHeaderSize(bool is_bind, bool is_ipv6) {
    81	  return is_bind ? (sizeof(uint8_t) +  // IP Version.
    82	                    (is_ipv6 ? quiche::QuicheIpAddress::kIPv6AddressSize
    83	                             : quiche::QuicheIpAddress::kIPv4AddressSize) +
    84	                    sizeof(uint16_t))  // Port.
    85	                 : 0;
    86	}
    87	
    88	constexpr size_t GetMaxDatagramHeaderSize(bool is_bind, bool is_ipv6) {
    89	  return is_bind ? (sizeof(MasqueServerSession::ContextId) +
    90	                    GetMaxIpHeaderSize(is_bind, is_ipv6))
    91	                 : 1;
    92	}

    94	// RAII wrapper for QuicUdpSocketFd.
    95	class FdWrapper {
    96	 public:
    97	  // Takes ownership of |fd| and closes the file descriptor on destruction.
    98	  explicit FdWrapper(int address_family) {
    99	    QuicUdpSocketApi socket_api;
   100	    fd_ =
   101	        socket_api.Create(address_family,
   102	                          /*receive_buffer_size =*/kDefaultSocketReceiveBuffer,
   103	                          /*send_buffer_size =*/kDefaultSocketReceiveBuffer);
   104	  }
   105	
   106	  ~FdWrapper() {
   107	    if (fd_ == kQuicInvalidSocketFd) {
   108	      return;
   109	    }
   110	    QuicUdpSocketApi socket_api;
   111	    socket_api.Destroy(fd_);
   112	  }
   113	
   114	  // Hands ownership of the file descriptor to the caller.
   115	  QuicUdpSocketFd extract_fd() {
   116	    QuicUdpSocketFd fd = fd_;
   117	    fd_ = kQuicInvalidSocketFd;
   118	    return fd;
   119	  }
   120	
   121	  // Keeps ownership of the file descriptor.
   122	  QuicUdpSocketFd fd() { return fd_; }
   123	
   124	  // Disallow copy and move.
   125	  FdWrapper(const FdWrapper&) = delete;
   126	  FdWrapper(FdWrapper&&) = delete;
   127	  FdWrapper& operator=(const FdWrapper&) = delete;
   128	  FdWrapper& operator=(FdWrapper&&) = delete;
   129	
   130	 private:
   131	  QuicUdpSocketFd fd_;
   132	};

   134	std::unique_ptr<QuicBackendResponse> CreateBackendErrorResponse(
   135	    absl::string_view status, absl::string_view error_details) {
   136	  quiche::HttpHeaderBlock response_headers;
   137	  response_headers[":status"] = status;
   138	  response_headers["masque-debug-info"] = error_details;
   139	  auto response = std::make_unique<QuicBackendResponse>();
   140	  response->set_response_type(QuicBackendResponse::REGULAR_RESPONSE);
   141	  response->set_headers(std::move(response_headers));
   142	  return response;
   143	}

   147	MasqueServerSession::MasqueServerSession(
   148	    MasqueMode masque_mode, const QuicConfig& config,
   149	    const ParsedQuicVersionVector& supported_versions,
   150	    QuicConnection* connection, QuicSession::Visitor* visitor,
   151	    QuicEventLoop* event_loop, QuicCryptoServerStreamBase::Helper* helper,
   152	    const QuicCryptoServerConfig* crypto_config,
   153	    QuicCompressedCertsCache* compressed_certs_cache,
   154	    MasqueServerBackend* masque_server_backend)
   155	    : QuicSimpleServerSession(config, supported_versions, connection, visitor,
   156	                              helper, crypto_config, compressed_certs_cache,
   157	                              masque_server_backend),
   158	      masque_server_backend_(masque_server_backend),
   159	      event_loop_(event_loop),
   160	      masque_mode_(masque_mode) {
   161	  // Artificially increase the max packet length to ensure we can fit QUIC
   162	  // packets inside DATAGRAM frames.
   163	  // TODO(b/181606597) Remove this workaround once we use PMTUD.
   164	  connection->SetMaxPacketLength(kDefaultMaxPacketSizeForTunnels);
   165	
   166	  masque_server_backend_->RegisterBackendClient(this);
   167	  QUICHE_DCHECK_NE(event_loop_, nullptr);
   168	
   169	  // We don't currently use `masque_mode_` but will in the future. To silence
   170	  // clang's `-Wunused-private-field` warning for this when building QUICHE for
   171	  // Chrome, add a use of it here.
   172	  (void)masque_mode_;
   173	}

   183	void MasqueServerSession::OnConnectionClosed(
   184	    const QuicConnectionCloseFrame& frame, ConnectionCloseSource source) {
   185	  QuicSimpleServerSession::OnConnectionClosed(frame, source);
   186	  QUIC_DLOG(INFO) << "Closing connection for " << connection_id();
   187	  masque_server_backend_->RemoveBackendClient(this);
   188	  // Clearing this state will close all sockets.
   189	  connect_udp_server_states_.clear();
   190	}

   192	void MasqueServerSession::OnStreamClosed(QuicStreamId stream_id) {
   193	  connect_udp_server_states_.remove_if(
   194	      [stream_id](const ConnectUdpServerState& connect_udp) {
   195	        return connect_udp.stream()->id() == stream_id;
   196	      });
   197	  connect_ip_server_states_.remove_if(
   198	      [stream_id](const ConnectIpServerState& connect_ip) {
   199	        return connect_ip.stream()->id() == stream_id;
   200	      });
   201	  connect_ethernet_server_states_.remove_if(
   202	      [stream_id](const ConnectEthernetServerState& connect_ethernet) {
   203	        return connect_ethernet.stream()->id() == stream_id;
   204	      });
   205	
   206	  QuicSimpleServerSession::OnStreamClosed(stream_id);
   207	}

   209	std::unique_ptr<QuicBackendResponse>
   210	MasqueServerSession::MaybeCheckConcealedAuth(
   211	    const quiche::HttpHeaderBlock& request_headers, absl::string_view authority,
   212	    absl::string_view scheme,
   213	    QuicSimpleServerBackend::RequestHandler* request_handler) {
   214	  // TODO(dschinazi) Add command-line flag that makes this implementation
   215	  // probe-resistant by returning the usual failure instead of 401.
   216	  constexpr absl::string_view kConcealedAuthStatus = "401";
   217	  if (!masque_server_backend_->IsConcealedAuthEnabled()) {
   218	    return nullptr;
   219	  }
   220	  auto authorization_pair = request_headers.find("authorization");
   221	  if (authorization_pair == request_headers.end()) {
   222	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   223	                                      "Missing authorization header");
   224	  }
   225	  absl::string_view credentials = authorization_pair->second;
   226	  quiche::QuicheTextUtils::RemoveLeadingAndTrailingWhitespace(&credentials);
   227	  std::vector<absl::string_view> v =
   228	      absl::StrSplit(credentials, absl::MaxSplits(' ', 1));
   229	  if (v.size() != 2) {
   230	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   231	                                      "Authorization header missing space");
   232	  }
   233	  absl::string_view auth_scheme = v[0];
   234	  if (auth_scheme != "Concealed") {
   235	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   236	                                      "Unexpected auth scheme");
   237	  }
   238	  absl::string_view auth_parameters = v[1];
   239	  std::vector<absl::string_view> auth_parameters_split =
   240	      absl::StrSplit(auth_parameters, ',');
   241	  std::optional<std::string> key_id;
   242	  std::optional<std::string> header_public_key;
   243	  std::optional<std::string> proof;
   244	  std::optional<uint16_t> signature_scheme;
   245	  std::optional<std::string> verification;
   246	  for (absl::string_view auth_parameter : auth_parameters_split) {
   247	    std::vector<absl::string_view> auth_parameter_split =
   248	        absl::StrSplit(auth_parameter, absl::MaxSplits('=', 1));
   249	    if (auth_parameter_split.size() != 2) {
   250	      continue;
   251	    }
   252	    absl::string_view param_name = auth_parameter_split[0];
   253	    quiche::QuicheTextUtils::RemoveLeadingAndTrailingWhitespace(&param_name);
   254	    if (param_name.size() != 1) {
   255	      // All currently known authentication parameters are one character long.
   256	      continue;
   257	    }
   258	    absl::string_view param_value = auth_parameter_split[1];
   259	    quiche::QuicheTextUtils::RemoveLeadingAndTrailingWhitespace(&param_value);
   260	    std::string decoded_param;
   261	    switch (param_name[0]) {
   262	      case 'k': {
   263	        if (key_id.has_value()) {
   264	          return CreateBackendErrorResponse(kConcealedAuthStatus,
   265	                                            "Duplicate k");
   266	        }
   267	        if (!absl::WebSafeBase64Unescape(param_value, &decoded_param)) {
   268	          return CreateBackendErrorResponse(kConcealedAuthStatus,
   269	                                            "Failed to base64 decode k");
   270	        }
   271	        key_id = decoded_param;
   272	      } break;
   273	      case 'a': {
   274	        if (header_public_key.has_value()) {
   275	          return CreateBackendErrorResponse(kConcealedAuthStatus,
   276	                                            "Duplicate a");
   277	        }
   278	        if (!absl::WebSafeBase64Unescape(param_value, &decoded_param)) {
   279	          return CreateBackendErrorResponse(kConcealedAuthStatus,
   280	                                            "Failed to base64 decode a");
   281	        }
   282	        header_public_key = decoded_param;
   283	      } break;
   284	      case 'p': {
   285	        if (proof.has_value()) {
   286	          return CreateBackendErrorResponse(kConcealedAuthStatus,
   287	                                            "Duplicate p");
   288	        }
   289	        if (!absl::WebSafeBase64Unescape(param_value, &decoded_param)) {
   290	          return CreateBackendErrorResponse(kConcealedAuthStatus,
   291	                                            "Failed to base64 decode p");
   292	        }
   293	        proof = decoded_param;
   294	      } break;
   295	      case 's': {
   296	        if (signature_scheme.has_value()) {
   297	          return CreateBackendErrorResponse(kConcealedAuthStatus,
   298	                                            "Duplicate s");
   299	        }
   300	        int signature_scheme_int = 0;
   301	        if (!absl::SimpleAtoi(param_value, &signature_scheme_int) ||
   302	            signature_scheme_int < 0 ||
   303	            signature_scheme_int > std::numeric_limits<uint16_t>::max()) {
   304	          return CreateBackendErrorResponse(kConcealedAuthStatus,
   305	                                            "Failed to parse s");
   306	        }
   307	        signature_scheme = static_cast<uint16_t>(signature_scheme_int);
   308	      } break;
   309	      case 'v': {
   310	        if (verification.has_value()) {
   311	          return CreateBackendErrorResponse(kConcealedAuthStatus,
   312	                                            "Duplicate v");
   313	        }
   314	        if (!absl::WebSafeBase64Unescape(param_value, &decoded_param)) {
   315	          return CreateBackendErrorResponse(kConcealedAuthStatus,
   316	                                            "Failed to base64 decode v");
   317	        }
   318	        verification = decoded_param;
   319	      } break;
   320	    }
   321	  }
   322	  if (!key_id.has_value()) {
   323	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   324	                                      "Missing k auth parameter");
   325	  }
   326	  if (!header_public_key.has_value()) {
   327	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   328	                                      "Missing a auth parameter");
   329	  }
   330	  if (!proof.has_value()) {
   331	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   332	                                      "Missing p auth parameter");
   333	  }
   334	  if (!signature_scheme.has_value()) {
   335	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   336	                                      "Missing s auth parameter");
   337	  }
   338	  if (!verification.has_value()) {
   339	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   340	                                      "Missing v auth parameter");
   341	  }
   342	  uint8_t config_public_key[ED25519_PUBLIC_KEY_LEN];
   343	  if (!masque_server_backend_->GetConcealedAuthKeyForId(*key_id,
   344	                                                        config_public_key)) {
   345	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   346	                                      "Unexpected key id");
   347	  }
   348	  if (*header_public_key !=
   349	      std::string(reinterpret_cast<const char*>(config_public_key),
   350	                  sizeof(config_public_key))) {
   351	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   352	                                      "Unexpected public key in header");
   353	  }
   354	  std::string realm = "";
   355	  QuicUrl url(absl::StrCat(scheme, "://", authority, "/"));
   356	  std::optional<std::string> key_exporter_context = ComputeConcealedAuthContext(
   357	      kEd25519SignatureScheme, *key_id, *header_public_key, scheme, url.host(),
   358	      url.port(), realm);
   359	  if (!key_exporter_context.has_value()) {
   360	    return CreateBackendErrorResponse(
   361	        "500", "Failed to generate key exporter context");
   362	  }
   363	  QUIC_DVLOG(1) << "key_exporter_context: "
   364	                << absl::WebSafeBase64Escape(*key_exporter_context);
   365	  QUICHE_DCHECK(!key_exporter_context->empty());
   366	  std::string key_exporter_output;
   367	  if (!GetMutableCryptoStream()->ExportKeyingMaterial(
   368	          kConcealedAuthLabel, *key_exporter_context,
   369	          kConcealedAuthExporterSize, &key_exporter_output)) {
   370	    return CreateBackendErrorResponse("500", "Key exporter failed");
   371	  }
   372	  QUICHE_CHECK_EQ(key_exporter_output.size(), kConcealedAuthExporterSize);
   373	  std::string signature_input =
   374	      key_exporter_output.substr(0, kConcealedAuthSignatureInputSize);
   375	  QUIC_DVLOG(1) << "signature_input: "
   376	                << absl::WebSafeBase64Escape(signature_input);
   377	  std::string expected_verification = key_exporter_output.substr(
   378	      kConcealedAuthSignatureInputSize, kConcealedAuthVerificationSize);
   379	  if (verification != expected_verification) {
   380	    return CreateBackendErrorResponse(
   381	        kConcealedAuthStatus,
   382	        absl::StrCat("Unexpected verification, expected ",
   383	                     absl::WebSafeBase64Escape(expected_verification),
   384	                     " but got ", absl::WebSafeBase64Escape(*verification),
   385	                     " - key exporter context was ",
   386	                     absl::WebSafeBase64Escape(*key_exporter_context)));
   387	  }
   388	  std::string data_covered_by_signature =
   389	      ConcealedAuthDataCoveredBySignature(signature_input);
   390	  QUIC_DVLOG(1) << "data_covered_by_signature: "
   391	                << absl::WebSafeBase64Escape(data_covered_by_signature);
   392	  if (*signature_scheme != kEd25519SignatureScheme) {
   393	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   394	                                      "Unexpected signature scheme");
   395	  }
   396	  if (proof->size() != ED25519_SIGNATURE_LEN) {
   397	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   398	                                      "Unexpected proof length");
   399	  }
   400	  if (ED25519_verify(
   401	          reinterpret_cast<const uint8_t*>(data_covered_by_signature.data()),
   402	          data_covered_by_signature.size(),
   403	          reinterpret_cast<const uint8_t*>(proof->data()),
   404	          config_public_key) != 1) {
   405	    return CreateBackendErrorResponse(kConcealedAuthStatus,
   406	                                      "Signature failed to validate");
   407	  }
   408	  QUIC_LOG(INFO) << "Successfully validated signature auth for stream ID "
   409	                 << request_handler->stream_id();
   410	  return nullptr;
   411	}

   413	std::unique_ptr<QuicBackendResponse> MasqueServerSession::HandleMasqueRequest(
   414	    const quiche::HttpHeaderBlock& request_headers,
   415	    QuicSimpleServerBackend::RequestHandler* request_handler) {
   416	  auto connect_udp_bind_pair = request_headers.find("connect-udp-bind");
   417	  bool has_bind_header = false;
   418	  if (connect_udp_bind_pair != request_headers.end()) {
   419	    absl::string_view connect_udp_bind_value = connect_udp_bind_pair->second;
   420	    std::vector<absl::string_view> params =
   421	        absl::StrSplit(connect_udp_bind_value, ';', absl::SkipEmpty());
   422	    if (!params.empty()) {
   423	      quiche::QuicheTextUtils::RemoveLeadingAndTrailingWhitespace(&params[0]);
   424	    }
   425	
   426	    if (!params.empty() && params[0] == "?1") {
   427	      has_bind_header = true;
   428	    } else if (params.empty() || params[0] != "?0") {
   429	      // Bind doesn't have a true or false value.
   430	      QUIC_DLOG(ERROR) << "Bad value for connect-udp-bind header: "
   431	                       << connect_udp_bind_value;
   432	      return CreateBackendErrorResponse(
   433	          "400", absl::StrCat("Bad value for connect-udp-bind header:",
   434	                              connect_udp_bind_value));
   435	    }
   436	  }
   437	
   438	  // Authority.
   439	  auto authority_pair = request_headers.find(":authority");
   440	  if (authority_pair == request_headers.end()) {
   441	    QUIC_DLOG(ERROR) << "MASQUE request is missing :authority";
   442	    return CreateBackendErrorResponse("400", "Missing :authority");
   443	  }
   444	  absl::string_view authority = authority_pair->second;
   445	  // Scheme.
   446	  auto scheme_pair = request_headers.find(":scheme");
   447	  if (scheme_pair == request_headers.end()) {
   448	    QUIC_DLOG(ERROR) << "MASQUE request is missing :scheme";
   449	    return CreateBackendErrorResponse("400", "Missing :scheme");
   450	  }
   451	  absl::string_view scheme = scheme_pair->second;
   452	  if (scheme.empty()) {
   453	    return CreateBackendErrorResponse("400", "Empty scheme");
   454	  }
   455	  // Concealed authentication.
   456	  auto concealed_auth_reply = MaybeCheckConcealedAuth(
   457	      request_headers, authority, scheme, request_handler);
   458	  if (concealed_auth_reply) {
   459	    return concealed_auth_reply;
   460	  }
   461	  // Path.
   462	  auto path_pair = request_headers.find(":path");
   463	  if (path_pair == request_headers.end()) {
   464	    QUIC_DLOG(ERROR) << "MASQUE request is missing :path";
   465	    return CreateBackendErrorResponse("400", "Missing :path");
   466	  }
   467	  absl::string_view path = path_pair->second;
   468	  if (path.empty()) {
   469	    QUIC_DLOG(ERROR) << "MASQUE request with empty path";
   470	    return CreateBackendErrorResponse("400", "Empty path");
   471	  }
   472	  // Method.
   473	  auto method_pair = request_headers.find(":method");
   474	  if (method_pair == request_headers.end()) {
   475	    QUIC_DLOG(ERROR) << "MASQUE request is missing :method";
   476	    return CreateBackendErrorResponse("400", "Missing :method");
   477	  }
   478	  absl::string_view method = method_pair->second;
   479	  if (method != "CONNECT") {
   480	    QUIC_DLOG(ERROR) << "MASQUE request with bad method \"" << method << "\"";
   481	    if (masque_server_backend_->IsConcealedAuthOnAllRequests()) {
   482	      return nullptr;
   483	    } else {
   484	      return CreateBackendErrorResponse("400", "Bad method");
   485	    }
   486	  }
   487	  // Protocol.
   488	  auto protocol_pair = request_headers.find(":protocol");
   489	  if (protocol_pair == request_headers.end()) {
   490	    QUIC_DLOG(ERROR) << "MASQUE request is missing :protocol";
   491	    if (masque_server_backend_->IsConcealedAuthOnAllRequests()) {
   492	      return nullptr;
   493	    } else {
   494	      return CreateBackendErrorResponse("400", "Missing :protocol");
   495	    }
   496	  }
   497	  absl::string_view protocol = protocol_pair->second;
   498	  if (protocol != "connect-udp" && protocol != "connect-ip" &&
   499	      protocol != "connect-ethernet") {
   500	    QUIC_DLOG(ERROR) << "MASQUE request with bad protocol \"" << protocol
   501	                     << "\"";
   502	    if (masque_server_backend_->IsConcealedAuthOnAllRequests()) {
   503	      return nullptr;
   504	    } else {
   505	      return CreateBackendErrorResponse("400", "Bad protocol");
   506	    }
   507	  }
   508	
   509	  if (protocol == "connect-ip") {
   510	    QuicSpdyStream* stream = absl::down_cast<QuicSpdyStream*>(
   511	        GetActiveStream(request_handler->stream_id()));
   512	    if (stream == nullptr) {
   513	      QUIC_BUG(bad masque server stream type)
   514	          << "Unexpected stream type for stream ID "
   515	          << request_handler->stream_id();
   516	      return CreateBackendErrorResponse("500", "Bad stream type");
   517	    }
   518	    QuicIpAddress client_ip = masque_server_backend_->GetNextClientIpAddress();
   519	    QUIC_DLOG(INFO) << "Using client IP " << client_ip.ToString()
   520	                    << " for CONNECT-IP stream ID "
   521	                    << request_handler->stream_id();
   522	    int fd = CreateTunInterface(client_ip);
   523	    if (fd < 0) {
   524	      QUIC_LOG(ERROR) << "Failed to create TUN interface for stream ID "
   525	                      << request_handler->stream_id();
   526	      return CreateBackendErrorResponse("500",
   527	                                        "Failed to create TUN interface");
   528	    }
   529	    if (!event_loop_->RegisterSocket(fd, kSocketEventReadable, this)) {
   530	      QUIC_DLOG(ERROR) << "Failed to register TUN fd with the event loop";
   531	      close(fd);
   532	      return CreateBackendErrorResponse("500", "Registering TUN socket failed");
   533	    }
   534	    connect_ip_server_states_.push_back(
   535	        ConnectIpServerState(client_ip, stream, fd, this));
   536	
   537	    quiche::HttpHeaderBlock response_headers;
   538	    response_headers[":status"] = "200";
   539	    auto response = std::make_unique<QuicBackendResponse>();
   540	    response->set_response_type(QuicBackendResponse::INCOMPLETE_RESPONSE);
   541	    response->set_headers(std::move(response_headers));
   542	    response->set_body("");
   543	
   544	    return response;
   545	  }
   546	  if (protocol == "connect-ethernet") {
   547	    QuicSpdyStream* stream = static_cast<QuicSpdyStream*>(
   548	        GetActiveStream(request_handler->stream_id()));
   549	    if (stream == nullptr) {
   550	      QUIC_BUG(bad masque server stream type)
   551	          << "Unexpected stream type for stream ID "
   552	          << request_handler->stream_id();
   553	      return CreateBackendErrorResponse("500", "Bad stream type");
   554	    }
   555	    int fd = CreateTapInterface();
   556	    if (fd < 0) {
   557	      QUIC_LOG(ERROR) << "Failed to create TAP interface for stream ID "
   558	                      << request_handler->stream_id();
   559	      return CreateBackendErrorResponse("500",
   560	                                        "Failed to create TAP interface");
   561	    }
   562	    if (!event_loop_->RegisterSocket(fd, kSocketEventReadable, this)) {
   563	      QUIC_DLOG(ERROR) << "Failed to register TAP fd with the event loop";
   564	      close(fd);
   565	      return CreateBackendErrorResponse("500", "Registering TAP socket failed");
   566	    }
   567	    connect_ethernet_server_states_.push_back(
   568	        ConnectEthernetServerState(stream, fd, this));
   569	
   570	    quiche::HttpHeaderBlock response_headers;
   571	    response_headers[":status"] = "200";
   572	    auto response = std::make_unique<QuicBackendResponse>();
   573	    response->set_response_type(QuicBackendResponse::INCOMPLETE_RESPONSE);
   574	    response->set_headers(std::move(response_headers));
   575	    response->set_body("");
   576	
   577	    return response;
   578	  }
   579	
   580	  // Extract target host and port from path using default template.
   581	  std::vector<absl::string_view> path_split = absl::StrSplit(path, '/');
   582	  if (path_split.size() != 7 || !path_split[0].empty() ||
   583	      path_split[1] != ".well-known" || path_split[2] != "masque" ||
   584	      path_split[3] != "udp" || path_split[4].empty() ||
   585	      path_split[5].empty() || !path_split[6].empty()) {
   586	    QUIC_DLOG(ERROR) << "MASQUE request with bad path \"" << path << "\"";
   587	    return CreateBackendErrorResponse("400", "Bad path");
   588	  }
   589	  std::optional<std::string> host = quiche::AsciiUrlDecode(path_split[4]);
   590	  if (!host.has_value()) {
   591	    QUIC_DLOG(ERROR) << "Failed to decode host \"" << path_split[4] << "\"";
   592	    return CreateBackendErrorResponse("500", "Failed to decode host");
   593	  }
   594	  std::optional<std::string> port = quiche::AsciiUrlDecode(path_split[5]);
   595	  if (!port.has_value()) {
   596	    QUIC_DLOG(ERROR) << "Failed to decode port \"" << path_split[5] << "\"";
   597	    return CreateBackendErrorResponse("500", "Failed to decode port");
   598	  }
   599	
   600	  QuicSocketAddress target_server_address;
   601	  if (has_bind_header && host == "*" && port == "*") {
   602	    // This is a valid connect-udp-bind request.
   603	    target_server_address = QuicSocketAddress();
   604	  } else if (has_bind_header) {
   605	    QUIC_DLOG(ERROR)
   606	        << "Bad value for target when connect-udp-bind is present: host:"
   607	        << host.value() << " port: " << port.value();
   608	    return CreateBackendErrorResponse(
   609	        "400",
   610	        absl::StrCat(
   611	            "Bad value for target when connect-udp-bind is present: host:",
   612	            host.value(), " port: ", port.value()));
   613	  } else if (host == "*" || port == "*") {
   614	    // Error.
   615	    QUIC_DLOG(ERROR)
   616	        << "Bad value for target when connect-udp-bind is absent: host:"
   617	        << host.value() << " port: " << port.value();
   618	    return CreateBackendErrorResponse(
   619	        "400",
   620	        absl::StrCat(
   621	            "Bad value for target when connect-udp-bind is absent: host:",
   622	            host.value(), " port: ", port.value()));
   623	
   624	  } else {
   625	    // Non-bind connect udp.
   626	
   627	    // Perform DNS resolution.
   628	
   629	    target_server_address =
   630	        quic::tools::LookupAddress(AF_UNSPEC, host.value(), port.value());
   631	
   632	    if (!target_server_address.IsInitialized()) {
   633	      QUIC_DLOG(ERROR) << "Failed to resolve " << authority << ": "
   634	                       << host.value() << ":" << port.value();
   635	      return CreateBackendErrorResponse("500", "DNS resolution failed");
   636	    }
   637	  }
   638	
   639	  QUIC_DLOG(INFO) << "Got CONNECT_UDP request on stream ID "
   640	                  << request_handler->stream_id() << " target_server_address=\""
   641	                  << target_server_address << "\"";
   642	  bool use_ipv6 = false;
   643	  if (has_bind_header) {
   644	    use_ipv6 = true;  // V4 Mapped V6.
   645	  } else {
   646	    // Match the target server address family.
   647	    use_ipv6 = target_server_address.host().AddressFamilyToInt() == AF_INET6;
   648	  }
   649	
   650	  QuicSpdyStream* stream = absl::down_cast<QuicSpdyStream*>(
   651	      GetActiveStream(request_handler->stream_id()));
   652	  if (stream == nullptr) {
   653	    QUIC_BUG(bad masque server stream type)
   654	        << "Unexpected stream type for stream ID "
   655	        << request_handler->stream_id();
   656	    return CreateBackendErrorResponse("500", "Bad stream type");
   657	  }
   658	
   659	  FdWrapper fd_wrapper(use_ipv6 ? AF_INET6 : AF_INET);
   660	  if (fd_wrapper.fd() == kQuicInvalidSocketFd) {
   661	    QUIC_DLOG(ERROR) << "Socket creation failed";
   662	    return CreateBackendErrorResponse("500", "Socket creation failed");
   663	  }
   664	  QuicSocketAddress empty_address(
   665	      (use_ipv6 ? QuicIpAddress::Any6() : QuicIpAddress::Any4()), 0);
   666	  QuicUdpSocketApi socket_api;
   667	  if (!socket_api.Bind(fd_wrapper.fd(), empty_address)) {
   668	    QUIC_DLOG(ERROR) << "Socket bind failed";
   669	    return CreateBackendErrorResponse("500", "Socket bind failed");
   670	  }
   671	  if (!event_loop_->RegisterSocket(fd_wrapper.fd(), kSocketEventReadable,
   672	                                   this)) {
   673	    QUIC_DLOG(ERROR) << "Failed to register socket with the event loop";
   674	    return CreateBackendErrorResponse("500", "Registering socket failed");
   675	  }
   676	  QuicSocketAddress configured_address;
   677	  configured_address.FromSocket(fd_wrapper.fd());
   678	  QUIC_LOG(INFO) << "Socket created, listening at port "
   679	                 << configured_address.port();
   680	
   681	  connect_udp_server_states_.push_back(ConnectUdpServerState(
   682	      stream, target_server_address, fd_wrapper.extract_fd(), this,
   683	      /*is_bind = */ has_bind_header));
   684	
   685	  quiche::HttpHeaderBlock response_headers;
   686	  response_headers[":status"] = "200";
   687	  if (has_bind_header) {
   688	    // For now the proxy public address is returned as
   689	    // localhost values.
   690	    // TODO(abhisinghx) : return the actual public address.
   691	    response_headers["proxy-public-address"] = configured_address.ToString();
   692	  }
   693	  auto response = std::make_unique<QuicBackendResponse>();
   694	  response->set_response_type(QuicBackendResponse::INCOMPLETE_RESPONSE);
   695	  response->set_headers(std::move(response_headers));
   696	  response->set_body("");
   697	
   698	  return response;
   699	}

   703	void MasqueServerSession::OnSocketEvent(QuicEventLoop* /*event_loop*/,
   704	                                        QuicUdpSocketFd fd,
   705	                                        QuicSocketEventMask events) {
   706	  if ((events & kSocketEventReadable) == 0) {
   707	    QUIC_DVLOG(1) << "Ignoring OnEvent fd " << fd << " event mask " << events;
   708	    return;
   709	  }
   710	
   711	  auto rearm = absl::MakeCleanup([&]() {
   712	    if (!event_loop_->SupportsEdgeTriggered()) {
   713	      if (!event_loop_->RearmSocket(fd, kSocketEventReadable)) {
   714	        QUIC_BUG(MasqueServerSession_OnSocketEvent_Rearm)
   715	            << "Failed to re-arm socket " << fd << " for reading";
   716	      }
   717	    }
   718	  });
   719	
   720	  if (!(HandleConnectUdpSocketEvent(fd, events) ||
   721	        HandleConnectIpSocketEvent(fd, events) ||
   722	        HandleConnectEthernetSocketEvent(fd, events))) {
   723	    QUIC_BUG(MasqueServerSession_OnSocketEvent_UnhandledEvent)
   724	        << "Got unexpected event mask " << events << " on unknown fd " << fd;
   725	    std::move(rearm).Cancel();
   726	  }
   727	}

   729	bool MasqueServerSession::HandleConnectUdpSocketEvent(
   730	    QuicUdpSocketFd fd, QuicSocketEventMask events) {
   731	  auto it = absl::c_find_if(connect_udp_server_states_,
   732	                            [fd](const ConnectUdpServerState& connect_udp) {
   733	                              return connect_udp.fd() == fd;
   734	                            });
   735	  if (it == connect_udp_server_states_.end()) {
   736	    return false;
   737	  }
   738	  bool is_bind = it->is_bind();
   739	  QuicSocketAddress expected_target_server_address =
   740	      it->target_server_address();
   741	
   742	  QUICHE_DCHECK(expected_target_server_address.IsInitialized() || is_bind);
   743	  QUIC_DVLOG(1) << "Received readable event on fd " << fd << " (mask " << events
   744	                << ") stream ID " << it->stream()->id()
   745	                << (is_bind ? " (bind)" : "") << " server "
   746	                << expected_target_server_address;
   747	  QuicUdpSocketApi socket_api;
   748	  QuicUdpPacketInfoBitMask packet_info_interested(
   749	      {QuicUdpPacketInfoBit::PEER_ADDRESS});
   750	
   751	  size_t ip_header_size = GetMaxIpHeaderSize(is_bind, /*is_ipv6=*/true);
   752	  constexpr size_t max_datagram_header_size =
   753	      GetMaxDatagramHeaderSize(/*is_bind=*/true, /*is_ipv6=*/true);
   754	
   755	  constexpr size_t packet_buffer_size =
   756	      max_datagram_header_size + kMaxIncomingPacketSize;
   757	
   758	  char packet_buffer[packet_buffer_size];
   759	
   760	  char control_buffer[kDefaultUdpPacketControlBufferSize];
   761	  while (true) {
   762	    QuicUdpSocketApi::ReadPacketResult read_result;
   763	    // Read, leaving space for context, ip version info etc.
   764	    read_result.packet_buffer = {packet_buffer + max_datagram_header_size,
   765	                                 packet_buffer_size - max_datagram_header_size};
   766	    read_result.control_buffer = {control_buffer, sizeof(control_buffer)};
   767	    socket_api.ReadPacket(fd, packet_info_interested, &read_result);
   768	    if (!read_result.ok) {
   769	      // Most likely there is nothing left to read, break out of read loop.
   770	      break;
   771	    }
   772	    if (!read_result.packet_info.HasValue(QuicUdpPacketInfoBit::PEER_ADDRESS)) {
   773	      QUIC_BUG(MasqueServerSession_HandleConnectUdpSocketEvent_MissingPeer)
   774	          << "Missing peer address when reading from fd " << fd;
   775	      continue;
   776	    }
   777	
   778	    // Normalize v6 mapped v4 to v4 if needed.
   779	    QuicSocketAddress peer_address =
   780	        read_result.packet_info.peer_address().Normalized();
   781	
   782	    ContextId bind_context_id = kInvalidContextId;
   783	    bool use_uncompressed_context = false;
   784	    if (is_bind) {
   785	      ContextId uncompressed_context = kInvalidContextId;
   786	      ContextId matching_context = kInvalidContextId;
   787	      // Now check if we either have a matching context or an uncompressed
   788	      // context.
   789	      // TODO(abhisinghx) : make this search more efficient.
   790	      for (auto& context_address_pair : it->bind_context_ip_map()) {
   791	        if (context_address_pair.second == peer_address) {
   792	          matching_context = context_address_pair.first;
   793	          break;
   794	        }
   795	        if (context_address_pair.second == quiche::QuicheSocketAddress()) {
   796	          uncompressed_context = context_address_pair.first;
   797	        }
   798	      }
   799	
   800	      if (matching_context != kInvalidContextId) {
   801	        bind_context_id = matching_context;
   802	      } else {
   803	        use_uncompressed_context = true;
   804	
   805	        if (uncompressed_context == kInvalidContextId) {
   806	          // We cannot accept this packet since we have no uncompressed context.
   807	          QUIC_DLOG(ERROR) << "Ignoring UDP packet on fd " << fd
   808	                           << " from unexpected server address " << peer_address
   809	                           << " (expected " << expected_target_server_address
   810	                           << ")";
   811	          continue;
   812	        }
   813	        bind_context_id = uncompressed_context;
   814	      }
   815	    } else if (peer_address != expected_target_server_address) {
   816	      QUIC_DLOG(ERROR) << "Ignoring UDP packet on fd " << fd
   817	                       << " from unexpected server address " << peer_address
   818	                       << " (expected " << expected_target_server_address
   819	                       << ")";
   820	      continue;
   821	    }
   822	    if (!connection()->connected()) {
   823	      QUIC_BUG(MasqueServerSession_HandleConnectUdpSocketEvent_ConnectionClosed)
   824	          << "Unexpected incoming UDP packet on fd " << fd << " from "
   825	          << expected_target_server_address
   826	          << " because MASQUE connection is closed";
   827	      return true;
   828	    }
   829	
   830	    size_t final_header_length;
   831	    char* udp_payload_read_pos = read_result.packet_buffer.buffer;
   832	    ip_header_size = GetMaxIpHeaderSize(is_bind, peer_address.host().IsIPv6());
   833	    final_header_length =
   834	        GetMaxDatagramHeaderSize(is_bind, peer_address.host().IsIPv6());
   835	
   836	    if (is_bind) {
   837	      // If compressed, ip header size is 0.
   838	      ip_header_size = use_uncompressed_context ? ip_header_size : 0;
   839	      size_t context_len = QuicDataWriter::GetVarInt62Len(bind_context_id);
   840	      final_header_length = ip_header_size + context_len;
   841	
   842	      // Now write the actual header.
   843	      QuicDataWriter writer(final_header_length,
   844	                            udp_payload_read_pos - final_header_length);
   845	      writer.WriteVarInt62(bind_context_id);
   846	      if (use_uncompressed_context) {
   847	        writer.WriteUInt8(
   848	            peer_address.host().AddressFamilyToInt() == AF_INET ? 4 : 6);
   849	        writer.WriteStringPiece(peer_address.host().ToPackedString());
   850	        writer.WriteUInt16(peer_address.port());
   851	      }
   852	    } else {
   853	      QuicDataWriter writer(final_header_length,
   854	                            udp_payload_read_pos - final_header_length);
   855	      writer.WriteVarInt62(/*context_id=*/0);
   856	    }
   857	
   858	    // The packet is valid, send it to the client in a DATAGRAM frame.
   859	    absl::string_view message(
   860	        udp_payload_read_pos - final_header_length,
   861	        read_result.packet_buffer.buffer_len + final_header_length);
   862	    DatagramStatus message_status = it->stream()->SendHttp3Datagram(message);
   863	    QUIC_DVLOG(1) << "Sent UDP packet from " << expected_target_server_address
   864	                  << " of length " << read_result.packet_buffer.buffer_len
   865	                  << " with stream ID " << it->stream()->id()
   866	                  << " and got message status "
   867	                  << DatagramStatusToString(message_status)
   868	                  << " message size with header: " << message.size();
   869	    QUIC_DVLOG(2) << "Contents of outgoing HTTP Datagram of length "
   870	                  << message.size() << ":" << std::endl
   871	                  << quiche::QuicheTextUtils::HexDump(message);
   872	  }
   873	  return true;
   874	}

   876	bool MasqueServerSession::HandleConnectIpSocketEvent(
   877	    QuicUdpSocketFd fd, QuicSocketEventMask events) {
   878	  auto it = absl::c_find_if(connect_ip_server_states_,
   879	                            [fd](const ConnectIpServerState& connect_ip) {
   880	                              return connect_ip.fd() == fd;
   881	                            });
   882	  if (it == connect_ip_server_states_.end()) {
   883	    return false;
   884	  }
   885	  QUIC_DVLOG(1) << "Received readable event on fd " << fd << " (mask " << events
   886	                << ") stream ID " << it->stream()->id();
   887	  char datagram[kMasqueIpPacketBufferSize];
   888	  datagram[0] = 0;  // Context ID.
   889	  while (true) {
   890	    ssize_t read_size = read(fd, datagram + 1, sizeof(datagram) - 1);
   891	    if (read_size < 0) {
   892	      break;
   893	    }
   894	    DatagramStatus message_status = it->stream()->SendHttp3Datagram(
   895	        absl::string_view(datagram, 1 + read_size));
   896	    QUIC_DVLOG(1) << "Encapsulated IP packet of length " << read_size
   897	                  << " with stream ID " << it->stream()->id()
   898	                  << " and got message status "
   899	                  << DatagramStatusToString(message_status);
   900	  }
   901	  return true;
   902	}

   915	                << ") stream ID " << it->stream()->id();
   916	  char datagram[kMasqueEthernetFrameBufferSize];
   917	  datagram[0] = 0;  // Context ID.
   918	  while (true) {
   919	    ssize_t read_size = read(fd, datagram + 1, sizeof(datagram) - 1);
   920	    if (read_size < 0) {
   921	      break;
   922	    }

   924	        absl::string_view(datagram, 1 + read_size));
   925	    QUIC_DVLOG(1) << "Encapsulated Ethernet frame of length " << read_size
   926	                  << " with stream ID " << it->stream()->id()
   927	                  << " and got message status "
   928	                  << DatagramStatusToString(message_status);
   929	  }
   930	  return true;

   946	MasqueServerSession::ConnectUdpServerState::ConnectUdpServerState(
   947	    QuicSpdyStream* stream, const QuicSocketAddress& target_server_address,
   948	    QuicUdpSocketFd fd, MasqueServerSession* masque_session, bool is_bind)
   949	    : stream_(stream),
   950	      target_server_address_(target_server_address),
   951	      fd_(fd),
   952	      masque_session_(masque_session),
   953	      is_bind_(is_bind) {
   954	  QUICHE_DCHECK_NE(fd_, kQuicInvalidSocketFd);
   955	  QUICHE_DCHECK_NE(masque_session_, nullptr);
   956	  this->stream()->RegisterHttp3DatagramVisitor(this);
   957	  if (is_bind) {
   958	    this->stream()->RegisterConnectUdpBindVisitor(this);
   959	  }
   960	}

   962	MasqueServerSession::ConnectUdpServerState::~ConnectUdpServerState() {
   963	  QuicUdpSocketApi socket_api;
   964	  if (stream() != nullptr) {
   965	    stream()->UnregisterHttp3DatagramVisitor();
   966	    if (is_bind_) {
   967	      stream()->UnregisterConnectUdpBindVisitor();
   968	    }
   969	  }
   970	  if (fd_ == kQuicInvalidSocketFd) {
   971	    return;
   972	  }
   973	  QUIC_DLOG(INFO) << "Closing fd " << fd_;
   974	  if (!masque_session_->event_loop()->UnregisterSocket(fd_)) {
   975	    QUIC_DLOG(ERROR) << "Failed to unregister FD " << fd_;
   976	  }
   977	  socket_api.Destroy(fd_);
   978	}

   980	MasqueServerSession::ConnectUdpServerState::ConnectUdpServerState(

  1013	void MasqueServerSession::ConnectUdpServerState::OnHttp3Datagram(
  1014	    QuicStreamId stream_id, absl::string_view payload) {
  1015	  QUICHE_DCHECK_EQ(stream_id, stream()->id());
  1016	  QuicDataReader reader(payload);
  1017	  ContextId context_id;
  1018	  if (!reader.ReadVarInt62(&context_id)) {
  1019	    QUIC_DLOG(ERROR) << "Failed to read context ID";
  1020	    return;
  1021	  }
  1022	  QuicSocketAddress target_address;
  1023	  if (!is_bind_ && context_id != 0) {
  1024	    QUIC_DLOG(ERROR) << "Ignoring HTTP Datagram with unexpected context ID "
  1025	                     << context_id;
  1026	    return;
  1027	  } else if (is_bind_) {
  1028	    auto it = bind_context_ip_map_.find(context_id);
  1029	    if (it == bind_context_ip_map_.end()) {
  1030	      QUIC_DLOG(ERROR) << "Context ID " << context_id
  1031	                       << " from datagram not found";
  1032	      return;
  1033	    }
  1034	    // Use context id to determine target.
  1035	    target_address = it->second;
  1036	  }
  1037	  if (!is_bind_) {
  1038	    target_address = target_server_address_;
  1039	  } else {
  1040	    QUIC_DVLOG(1) << "CONNECT-UDP-BIND datagram from client";
  1041	    // CONNECT-UDP-BIND.
  1042	    if (target_address.host().address_family() == IpAddressFamily::IP_UNSPEC) {
  1043	      // Parse IP and port.
  1044	      uint8_t ip_version;
  1045	      if (!reader.ReadUInt8(&ip_version)) {
  1046	        QUIC_DLOG(ERROR) << "Failed to read IP version";
  1047	        return;
  1048	      }
  1049	      if (ip_version != 4 && ip_version != 6) {
  1050	        QUIC_DLOG(ERROR) << "Invalid IP version " << ip_version;
  1051	        return;
  1052	      }
  1053	      size_t address_size = ip_version == 4
  1054	                                ? quiche::QuicheIpAddress::kIPv4AddressSize
  1055	                                : quiche::QuicheIpAddress::kIPv6AddressSize;
  1056	      absl::string_view ip_address_bytes;
  1057	      if (!reader.ReadStringPiece(&ip_address_bytes, address_size)) {
  1058	        QUIC_DLOG(ERROR) << "Failed to read IP address";
  1059	        return;
  1060	      }
  1061	      quiche::QuicheIpAddress ip_address;
  1062	      if (!ip_address.FromPackedString(ip_address_bytes.data(),
  1063	                                       ip_address_bytes.size())) {
  1064	        QUIC_DLOG(ERROR) << "Failed to parse IP address";
  1065	        return;
  1066	      }
  1067	      uint16_t port;
  1068	      if (!reader.ReadUInt16(&port)) {
  1069	        QUIC_DLOG(ERROR) << "Failed to read port";
  1070	        return;
  1071	      }
  1072	      target_address = QuicSocketAddress(ip_address, port);
  1073	    }
  1074	  }
  1075	
  1076	  absl::string_view http_payload = reader.ReadRemainingPayload();
  1077	  QuicUdpSocketApi socket_api;
  1078	  QuicUdpPacketInfo packet_info;
  1079	  packet_info.SetPeerAddress(target_address);
  1080	  WriteResult write_result = socket_api.WritePacket(
  1081	      fd_, http_payload.data(), http_payload.length(), packet_info);
  1082	  QUIC_DVLOG(1) << "Wrote packet of length " << http_payload.length() << " to "
  1083	                << target_address << " with result " << write_result;
  1084	  QUIC_DVLOG(2) << "Contents of outgoing UDP packet of length "
  1085	                << http_payload.length() << ":" << std::endl
  1086	                << quiche::QuicheTextUtils::HexDump(http_payload);
  1087	}

  1089	bool MasqueServerSession::ConnectUdpServerState::OnCompressionAssignCapsule(
  1090	    const quiche::CompressionAssignCapsule& capsule) {
  1091	  if (bind_context_ip_map_.contains(capsule.context_id)) {
  1092	    QUIC_DLOG(ERROR) << "Context ID " << capsule.context_id
  1093	                     << " from Compression Assign already exists";
  1094	    return false;
  1095	  } else {
  1096	    bind_context_ip_map_[capsule.context_id] = capsule.ip_address_port;
  1097	  }
  1098	  // Send acknowledgement.
  1099	  Capsule compression_assign_capsule = Capsule::CompressionAssign();
  1100	  compression_assign_capsule.compression_assign_capsule().context_id =
  1101	      capsule.context_id;
  1102	  compression_assign_capsule.compression_assign_capsule().ip_address_port =
  1103	      capsule.ip_address_port;
  1104	  stream()->WriteCapsule(compression_assign_capsule);
  1105	  // TODO(abhisinghx) : Compression Request Strategy for Server.
  1106	  return true;
  1107	}

  1109	bool MasqueServerSession::ConnectUdpServerState::OnCompressionCloseCapsule(
  1110	    const quiche::CompressionCloseCapsule& capsule) {
  1111	  auto it = bind_context_ip_map_.find(capsule.context_id);
  1112	  if (it != bind_context_ip_map_.end()) {
  1113	    bind_context_ip_map_.erase(it);
  1114	  } else {
  1115	    QUIC_DLOG(ERROR) << "Context ID " << capsule.context_id
  1116	                     << " from Compression Close not found";
  1117	    return false;
  1118	  }
  1119	  // Send acknowledgement.
  1120	  Capsule compression_close_capsule = Capsule::CompressionClose();
  1121	  compression_close_capsule.compression_close_capsule().context_id =
  1122	      capsule.context_id;
  1123	  stream()->WriteCapsule(compression_close_capsule);
  1124	  return true;
  1125	}

  1261	void MasqueServerSession::ConnectIpServerState::OnHeadersWritten() {
  1262	  QUICHE_DCHECK(client_ip_.IsIPv4()) << client_ip_.ToString();
  1263	  Capsule address_assign_capsule = Capsule::AddressAssign();
  1264	  PrefixWithId assigned_address;
  1265	  assigned_address.ip_prefix = quiche::QuicheIpPrefix(client_ip_, 32);
  1266	  assigned_address.request_id = 0;
  1267	  address_assign_capsule.address_assign_capsule().assigned_addresses.push_back(
  1268	      assigned_address);
  1269	  stream()->WriteCapsule(address_assign_capsule);
  1270	  IpAddressRange default_route;
  1271	  default_route.start_ip_address.FromString("0.0.0.0");
  1272	  default_route.end_ip_address.FromString("255.255.255.255");
  1273	  default_route.ip_protocol = 0;
  1274	  Capsule route_advertisement = Capsule::RouteAdvertisement();
  1275	  route_advertisement.route_advertisement_capsule().ip_address_ranges.push_back(
  1276	      default_route);
  1277	  stream()->WriteCapsule(route_advertisement);
  1278	
  1279	  if (masque_session_->masque_server_backend()->dns_assign_capsule()) {
  1280	    WriteDnsAssignCapsule(
  1281	        stream(),
  1282	        *masque_session_->masque_server_backend()->dns_assign_capsule());
  1283	  }
  1284	  if (masque_session_->masque_server_backend()->pref64_capsule()) {
  1285	    WritePref64Capsule(
  1286	        stream(), *masque_session_->masque_server_backend()->pref64_capsule());
  1287	  }
  1288	}
