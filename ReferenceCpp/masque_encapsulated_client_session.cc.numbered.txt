    58	void MasqueEncapsulatedClientSession::ProcessPacket(
    59	    absl::string_view packet, QuicSocketAddress server_address) {
    60	  QuicTime now = connection()->clock()->ApproximateNow();
    61	  QuicReceivedPacket received_packet(packet.data(), packet.length(), now);
    62	  connection()->ProcessUdpPacket(connection()->self_address(), server_address,
    63	                                 received_packet);
    64	}

    68	    ConnectionCloseBehavior connection_close_behavior) {
    69	  connection()->CloseConnection(error, details, connection_close_behavior);
    70	}

    78	void MasqueEncapsulatedClientSession::ProcessIpPacket(
    79	    absl::string_view packet) {
    80	  quiche::QuicheDataReader reader(packet);
    81	  uint8_t first_byte;
    82	  if (!reader.ReadUInt8(&first_byte)) {
    83	    QUIC_DLOG(ERROR) << "Dropping empty CONNECT-IP packet";
    84	    return;
    85	  }
    86	  const uint8_t ip_version = first_byte >> 4;
    87	  quiche::QuicheIpAddress server_ip;
    88	  if (ip_version == 6) {
    89	    if (!reader.Seek(5)) {
    90	      QUICHE_DLOG(ERROR) << "Failed to seek CONNECT-IP IPv6 start\n"
    91	                         << quiche::QuicheTextUtils::HexDump(packet);
    92	      return;
    93	    }
    94	    uint8_t next_header = 0;
    95	    if (!reader.ReadUInt8(&next_header)) {
    96	      QUICHE_DLOG(ERROR) << "Failed to read CONNECT-IP next header\n"
    97	                         << quiche::QuicheTextUtils::HexDump(packet);
    98	      return;
    99	    }
   100	    if (next_header != 17) {
   101	      // Note that this drops packets with IPv6 extension headers, since we
   102	      // do not expect to see them in practice.
   103	      QUIC_DLOG(ERROR)
   104	          << "Dropping CONNECT-IP packet with unexpected next header "
   105	          << static_cast<int>(next_header) << "\n"
   106	          << quiche::QuicheTextUtils::HexDump(packet);
   107	      return;
   108	    }
   109	    if (!reader.Seek(1)) {
   110	      QUICHE_DLOG(ERROR) << "Failed to seek CONNECT-IP hop limit\n"
   111	                         << quiche::QuicheTextUtils::HexDump(packet);
   112	      return;
   113	    }
   114	    absl::string_view source_ip;
   115	    if (!reader.ReadStringPiece(&source_ip, 16)) {
   116	      QUICHE_DLOG(ERROR) << "Failed to read CONNECT-IP source IPv6\n"
   117	                         << quiche::QuicheTextUtils::HexDump(packet);
   118	      return;
   119	    }
   120	    server_ip.FromPackedString(source_ip.data(), source_ip.length());
   121	    if (!reader.Seek(16)) {
   122	      QUICHE_DLOG(ERROR) << "Failed to seek CONNECT-IP destination IPv6\n"
   123	                         << quiche::QuicheTextUtils::HexDump(packet);
   124	      return;
   125	    }
   126	  } else if (ip_version == 4) {
   127	    uint8_t ihl = first_byte & 0xF;
   128	    if (ihl < 5) {
   129	      QUICHE_DLOG(ERROR) << "Dropping CONNECT-IP packet with invalid IHL "
   130	                         << static_cast<int>(ihl) << "\n"
   131	                         << quiche::QuicheTextUtils::HexDump(packet);
   132	      return;
   133	    }
   134	    if (!reader.Seek(8)) {
   135	      QUICHE_DLOG(ERROR) << "Failed to seek CONNECT-IP IPv4 start\n"
   136	                         << quiche::QuicheTextUtils::HexDump(packet);
   137	      return;
   138	    }
   139	    uint8_t ip_proto = 0;
   140	    if (!reader.ReadUInt8(&ip_proto)) {
   141	      QUICHE_DLOG(ERROR) << "Failed to read CONNECT-IP ip_proto\n"
   142	                         << quiche::QuicheTextUtils::HexDump(packet);
   143	      return;
   144	    }
   145	    if (ip_proto != 17) {
   146	      QUIC_DLOG(ERROR) << "Dropping CONNECT-IP packet with unexpected IP proto "
   147	                       << static_cast<int>(ip_proto) << "\n"
   148	                       << quiche::QuicheTextUtils::HexDump(packet);
   149	      return;
   150	    }
   151	    if (!reader.Seek(2)) {
   152	      QUICHE_DLOG(ERROR) << "Failed to seek CONNECT-IP IP checksum\n"
   153	                         << quiche::QuicheTextUtils::HexDump(packet);
   154	      return;
   155	    }
   156	    absl::string_view source_ip;
   157	    if (!reader.ReadStringPiece(&source_ip, 4)) {
   158	      QUICHE_DLOG(ERROR) << "Failed to read CONNECT-IP source IPv4\n"
   159	                         << quiche::QuicheTextUtils::HexDump(packet);
   160	      return;
   161	    }
   162	    server_ip.FromPackedString(source_ip.data(), source_ip.length());
   163	    if (!reader.Seek(4)) {
   164	      QUICHE_DLOG(ERROR) << "Failed to seek CONNECT-IP destination IPv4\n"
   165	                         << quiche::QuicheTextUtils::HexDump(packet);
   166	      return;
   167	    }
   168	    uint8_t ip_options_length = (ihl - 5) * 4;
   169	    if (!reader.Seek(ip_options_length)) {
   170	      QUICHE_DLOG(ERROR) << "Failed to seek CONNECT-IP IP options of length "
   171	                         << static_cast<int>(ip_options_length) << "\n"
   172	                         << quiche::QuicheTextUtils::HexDump(packet);
   173	      return;
   174	    }
   175	  } else {
   176	    QUIC_DLOG(ERROR) << "Dropping CONNECT-IP packet with unexpected IP version "
   177	                     << static_cast<int>(ip_version) << "\n"
   178	                     << quiche::QuicheTextUtils::HexDump(packet);
   179	    return;
   180	  }
   181	  // Parse UDP header.
   182	  uint16_t server_port;
   183	  if (!reader.ReadUInt16(&server_port)) {
   184	    QUICHE_DLOG(ERROR) << "Failed to read CONNECT-IP source port\n"
   185	                       << quiche::QuicheTextUtils::HexDump(packet);
   186	    return;
   187	  }
   188	  if (!reader.Seek(2)) {
   189	    QUICHE_DLOG(ERROR) << "Failed to seek CONNECT-IP destination port\n"
   190	                       << quiche::QuicheTextUtils::HexDump(packet);
   191	    return;
   192	  }
   193	  uint16_t udp_length;
   194	  if (!reader.ReadUInt16(&udp_length)) {
   195	    QUICHE_DLOG(ERROR) << "Failed to read CONNECT-IP UDP length\n"
   196	                       << quiche::QuicheTextUtils::HexDump(packet);
   197	    return;
   198	  }
   199	  if (udp_length < 8) {
   200	    QUICHE_DLOG(ERROR) << "Dropping CONNECT-IP packet with invalid UDP length "
   201	                       << udp_length << "\n"
   202	                       << quiche::QuicheTextUtils::HexDump(packet);
   203	    return;
   204	  }
   205	  if (!reader.Seek(2)) {
   206	    QUICHE_DLOG(ERROR) << "Failed to seek CONNECT-IP UDP checksum\n"
   207	                       << quiche::QuicheTextUtils::HexDump(packet);
   208	    return;
   209	  }
   210	  absl::string_view quic_packet;
   211	  if (!reader.ReadStringPiece(&quic_packet, udp_length - 8)) {
   212	    QUICHE_DLOG(ERROR) << "Failed to read CONNECT-IP UDP payload\n"
   213	                       << quiche::QuicheTextUtils::HexDump(packet);
   214	    return;
   215	  }
   216	  if (!reader.IsDoneReading()) {
   217	    QUICHE_DLOG(INFO) << "Received CONNECT-IP UDP packet with "
   218	                      << reader.BytesRemaining()
   219	                      << " extra bytes after payload\n"
   220	                      << quiche::QuicheTextUtils::HexDump(packet);
   221	  }
   222	  QUIC_DLOG(INFO) << "Received CONNECT-IP encapsulated packet of length "
   223	                  << quic_packet.size();
   224	  QuicTime now = connection()->clock()->ApproximateNow();
   225	  QuicReceivedPacket received_packet(quic_packet.data(), quic_packet.size(),
   226	                                     now);
   227	  QuicSocketAddress server_address = QuicSocketAddress(server_ip, server_port);
   228	  connection()->ProcessUdpPacket(connection()->self_address(), server_address,
   229	                                 received_packet);
   230	}

   238	bool MasqueEncapsulatedClientSession::OnAddressAssignCapsule(
   239	    const AddressAssignCapsule& capsule) {
   240	  QUIC_DLOG(INFO) << "Received capsule " << capsule.ToString();
   241	  for (auto assigned_address : capsule.assigned_addresses) {
   242	    if (assigned_address.ip_prefix.address().IsIPv4() &&
   243	        !local_v4_address_.IsInitialized()) {
   244	      QUIC_LOG(INFO)
   245	          << "MasqueEncapsulatedClientSession saving local IPv4 address "
   246	          << assigned_address.ip_prefix.address();
   247	      local_v4_address_ = assigned_address.ip_prefix.address();
   248	    } else if (assigned_address.ip_prefix.address().IsIPv6() &&
   249	               !local_v6_address_.IsInitialized()) {
   250	      QUIC_LOG(INFO)
   251	          << "MasqueEncapsulatedClientSession saving local IPv6 address "
   252	          << assigned_address.ip_prefix.address();
   253	      local_v6_address_ = assigned_address.ip_prefix.address();
   254	    }
   255	  }
   256	  return true;
   257	}
